cmake_minimum_required(VERSION 3.25)
project(native_ayu_data LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(native_ayu_database database.cpp ${TELEGRAM_ROOT}/lib_base/base/assertion.cpp)
target_include_directories(native_ayu_database PRIVATE ${TELEGRAM_ROOT}/SourceFiles ${TELEGRAM_ROOT}/lib_base ${TELEGRAM_ROOT}/lib_rpl ${TELEGRAM_ROOT}/lib_crl)
target_compile_options(native_ayu_database PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp -ffunction-sections -fdata-sections)
target_link_options(native_ayu_database PRIVATE -Wl,--gc-sections)
target_link_libraries(native_ayu_database PRIVATE Qt6::Core ${SQLITE_LIBRARY})
find_program(PYTHON NAMES python3 REQUIRED)
execute_process(COMMAND ${PYTHON} ${CMAKE_CURRENT_SOURCE_DIR}/extract_settings.py ${TELEGRAM_ROOT} ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_executable(native_ayu_settings settings.cpp ${TELEGRAM_ROOT}/lib_base/base/assertion.cpp)
target_include_directories(native_ayu_settings PRIVATE ${CMAKE_CURRENT_BINARY_DIR} ${TELEGRAM_ROOT}/SourceFiles ${TELEGRAM_ROOT}/lib_base ${TELEGRAM_ROOT}/lib_rpl ${TELEGRAM_ROOT}/lib_crl)
target_compile_options(native_ayu_settings PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp)
target_link_libraries(native_ayu_settings PRIVATE Qt6::Core)
target_sources(native_ayu_settings PRIVATE ${TELEGRAM_ROOT}/lib_base/base/integration.cpp)
target_compile_options(native_ayu_settings PRIVATE -ffunction-sections -fdata-sections)
target_link_options(native_ayu_settings PRIVATE -Wl,--gc-sections)
