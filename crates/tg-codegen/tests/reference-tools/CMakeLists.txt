cmake_minimum_required(VERSION 3.25)
project(original_codegen_only LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg)
set(root "${AYUGRAM_SOURCE}/Telegram")
file(GLOB common ${root}/codegen/codegen/common/*.cpp)
foreach(name style lang emoji)
  file(GLOB sources ${root}/codegen/codegen/${name}/*.cpp)
  add_executable(codegen_${name} ${sources} ${common})
  target_include_directories(codegen_${name} PRIVATE ${root}/codegen ${root}/lib_base)
  target_compile_options(codegen_${name} PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp)
  target_link_libraries(codegen_${name} PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
endforeach()
target_sources(codegen_style PRIVATE ${root}/lib_base/base/crc32hash.cpp)
file(GLOB_RECURSE objects ${CMAKE_BINARY_DIR}/CMakeFiles/codegen_style.dir/*.o)
list(FILTER objects EXCLUDE REGEX "/main.cpp.o$")
add_executable(style_ir_dump style_ir_dump.cpp ${objects})
target_include_directories(style_ir_dump PRIVATE ${root}/codegen ${root}/lib_base)
target_compile_options(style_ir_dump PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp)
target_link_libraries(style_ir_dump PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
file(GLOB_RECURSE emoji_objects ${CMAKE_BINARY_DIR}/CMakeFiles/codegen_emoji.dir/*.o)
list(FILTER emoji_objects EXCLUDE REGEX "/main.cpp.o$")
add_executable(emoji_ir_dump emoji_ir_dump.cpp ${emoji_objects})
target_include_directories(emoji_ir_dump PRIVATE ${root}/codegen ${root}/lib_base)
target_compile_options(emoji_ir_dump PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/prefix.hpp)
target_link_libraries(emoji_ir_dump PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
