cmake_minimum_required(VERSION 3.25)
project(native_auth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network)
add_executable(native_auth auth.cpp ${TELEGRAM_ROOT}/lib_tl/tl/tl_basic_types.cpp ${TELEGRAM_ROOT}/lib_base/base/assertion.cpp ${TELEGRAM_ROOT}/lib_base/base/integration.cpp)
target_include_directories(native_auth PRIVATE ${SCHEME_INCLUDE} ${TELEGRAM_ROOT}/SourceFiles ${TELEGRAM_ROOT}/lib_base ${TELEGRAM_ROOT}/lib_tl ${TELEGRAM_ROOT}/lib_rpl ${TELEGRAM_ROOT}/lib_crl)
target_compile_options(native_auth PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/../reference-tools/prefix.hpp -ffunction-sections -fdata-sections)
target_link_options(native_auth PRIVATE -Wl,--gc-sections)
target_link_libraries(native_auth PRIVATE Qt6::Core Qt6::Gui Qt6::Network)
target_sources(native_auth PRIVATE ${SCHEME_OBJECT})
